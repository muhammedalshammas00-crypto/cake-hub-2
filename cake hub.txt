        // Toggle Desktop / Phone Preview Mode (For Editing/Reviewing)
        function setPreviewMode(mode) {
            const body = document.body;
            const btnDesktop = document.getElementById('btn-desktop-view');
            const btnPhone = document.getElementById('btn-phone-view');
            const viewport = document.getElementById('app-viewport');

            if (mode === 'phone') {
                body.classList.add('phone-preview-mode');
                viewport.classList.add('custom-scrollbar');
                btnPhone.classList.add('bg-brand-yellow', 'text-zinc-900');
                btnPhone.classList.remove('text-zinc-300');
                btnDesktop.classList.remove('bg-brand-yellow', 'text-zinc-900');
                btnDesktop.classList.add('text-zinc-300');
                showToast('Phone View Enabled (390px Mobile Frame)');
            } else {
                body.classList.remove('phone-preview-mode');
                viewport.classList.remove('custom-scrollbar');
                btnDesktop.classList.add('bg-brand-yellow', 'text-zinc-900');
                btnDesktop.classList.remove('text-zinc-300');
                btnPhone.classList.remove('bg-brand-yellow', 'text-zinc-900');
                btnPhone.classList.add('text-zinc-300');
                showToast('Desktop View Enabled');
            }
        }

        // Safe icon initializer
        function refreshIcons() {
            try {
                if (window.lucide && typeof window.lucide.createIcons === 'function') {
                    window.lucide.createIcons();
                }
            } catch (e) {
                console.warn('Lucide icons skipped:', e);
            }
        }

        // Initialize App on load
        window.onload = function() {
            renderMenu();
            updateCartBadge();
            refreshIcons();
        };

        // Toast Notification Helper
        let toastTimeout;
        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-message');
            if (!toast || !toastMsg) return;
            toastMsg.textContent = message;
            toast.classList.remove('translate-y-24', 'opacity-0');
            clearTimeout(toastTimeout);
            toastTimeout = setTimeout(() => {
                toast.classList.add('translate-y-24', 'opacity-0');
            }, 2500);
        }

        // Render Menu Items Based on Category & Search
        function renderMenu() {
            const container = document.getElementById('menu-container');
            container.innerHTML = '';

            let hasResults = false;

            menuData.forEach(category => {
                if (currentFilter !== 'all' && category.id !== currentFilter) return;

                const filteredItems = category.items.filter(item =>
                    item.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
                    (item.desc && item.desc.toLowerCase().includes(searchQuery.toLowerCase()))
                );

                if (filteredItems.length === 0) return;
                hasResults = true;

                const section = document.createElement('section');
                section.id = `cat-${category.id}`;
                section.className = 'scroll-mt-32';

                let cardsHTML = '';
                filteredItems.forEach((item) => {
                    const priceKeys = Object.keys(item.prices);
                    const hasMultipleSizes = priceKeys.length > 1;
                    const defaultPrice = item.prices[priceKeys[0]];
                    const priceDisplay = hasMultipleSizes
                        ? priceKeys.map(k => `<span class="text-xs bg-amber-100 text-amber-900 px-2 py-0.5 rounded font-bold">${k}: ₹${item.prices[k]}</span>`).join(' ')
                        : `<span class="text-lg font-black text-zinc-900">₹${defaultPrice}</span>`;

                    const safeName = item.name.replace(/'/g, "\\'");
                    const itemImg = item.img || category.image;
                    const fallbackImg = `https://placehold.co/400x300/FFDE59/18181B?text=${encodeURIComponent(item.name)}`;

                    cardsHTML += `
                        <div class="bg-white rounded-2xl border border-amber-200/70 shadow-sm hover:shadow-xl transition-all duration-300 overflow-hidden flex flex-col justify-between group">
                            <div>
                                <div class="relative h-44 w-full overflow-hidden bg-amber-100">
                                    <img src="${itemImg}" onerror="this.onerror=null;this.src='${fallbackImg}';" alt="${item.name}" loading="lazy" class="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500">
                                    <div class="absolute inset-0 bg-gradient-to-t from-black/45 via-transparent to-transparent"></div>
                                    ${hasMultipleSizes ? `<span class="absolute top-2.5 right-2.5 bg-zinc-900/90 text-brand-yellow text-[10px] font-bold px-2.5 py-1 rounded-full uppercase tracking-wider shadow">Options</span>` : ''}
                                </div>
                                <div class="p-4">
                                    <h4 class="font-bold text-zinc-900 text-base group-hover:text-brand-red transition">${item.name}</h4>
                                    <p class="text-xs text-slate-500 mt-1 line-clamp-2">${item.desc || ''}</p>
                                </div>
                            </div>
                            <div class="px-4 pb-4 pt-2 border-t border-amber-100/80 flex items-center justify-between gap-2">
                                <div class="flex flex-wrap gap-1 items-center">
                                    ${priceDisplay}
                                </div>
                                <button onclick="handleItemAction('${category.id}', '${safeName}')" class="bg-brand-yellow hover:bg-zinc-900 text-zinc-900 hover:text-brand-yellow font-bold text-xs px-3.5 py-2 rounded-xl shadow-sm transition flex items-center gap-1 shrink-0 active:scale-95">
                                    <i data-lucide="plus" class="w-3.5 h-3.5"></i> Add
                                </button>
                            </div>
                        </div>
                    `;
                });

                section.innerHTML = `
                    <div class="relative rounded-2xl overflow-hidden mb-5 bg-zinc-900 text-white shadow-md">
                        <img src="${category.image}" onerror="this.style.display='none'" alt="${category.title}" class="absolute inset-0 w-full h-full object-cover opacity-35">
                        <div class="relative z-10 p-5 sm:p-6 flex flex-col sm:flex-row sm:items-center justify-between gap-2 bg-gradient-to-r from-zinc-950/90 via-zinc-900/75 to-transparent">
                            <div>
                                <h3 class="text-xl sm:text-2xl font-black text-brand-yellow tracking-tight">${category.title}</h3>
                                <p class="text-xs sm:text-sm text-slate-200 mt-0.5">${category.description}</p>
                            </div>
                            <span class="self-start sm:self-center text-xs font-bold bg-brand-yellow/20 text-brand-yellow border border-brand-yellow/40 px-3 py-1 rounded-full shrink-0">${filteredItems.length} Items</span>
                        </div>
                    </div>
                    <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-5">
                        ${cardsHTML}
                    </div>
                `;

                container.appendChild(section);
            });

            if (!hasResults) {
                container.innerHTML = `
                    <div class="text-center py-16 bg-white rounded-2xl border border-amber-200">
                        <i data-lucide="search-x" class="w-12 h-12 text-amber-400 mx-auto mb-3"></i>
                        <h3 class="text-lg font-bold text-zinc-800">No matching dishes found</h3>
                        <p class="text-sm text-slate-500 mt-1">Try searching for something else or view all items.</p>
                        <button onclick="resetFilters()" class="mt-4 bg-brand-yellow text-zinc-900 font-bold text-xs px-5 py-2.5 rounded-full shadow">Show All Menu</button>
                    </div>
                `;
            }

            refreshIcons();
        }

        function filterCategory(catId) {
            currentFilter = catId;
            document.querySelectorAll('.cat-pill').forEach(btn => {
                btn.className = 'cat-pill px-5 py-2.5 rounded-full text-sm font-semibold bg-white text-slate-700 hover:bg-slate-200 shadow-sm shrink-0 transition';
            });
            event.target.className = 'cat-pill px-5 py-2.5 rounded-full text-sm font-semibold bg-slate-900 text-white shadow-sm shrink-0 transition';
            renderMenu();
        }

        function resetFilters() {
            currentFilter = 'all';
            searchQuery = '';
            const searchInput = document.getElementById('menu-search');
            if (searchInput) searchInput.value = '';
            renderMenu();
        }

        function handleSearch(query) {
            searchQuery = query;
            renderMenu();
        }

        function handleItemAction(catId
